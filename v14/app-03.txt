      state.pinch.pendingZoom = Math.max(.5, Math.min(10, state.pinch.startZoom * scale));
      applyZoom(state.pinch.pendingZoom, false);
    };
    const finish = () => {
      if (!state.pinch.active) return;
      const z = state.pinch.pendingZoom; state.pinch.active = false;
      applyZoom(z, true); ui.previewZoomHud?.classList.remove('gesture');
    };
    surface.addEventListener('gesturestart', ev => { if (begin('gesture')) ev.preventDefault(); }, { passive: false });
    surface.addEventListener('gesturechange', ev => { if (state.pinch.active && state.pinch.source === 'gesture') { ev.preventDefault(); move(ev.scale || 1); } }, { passive: false });
    surface.addEventListener('gestureend', ev => { if (state.pinch.active && state.pinch.source === 'gesture') { ev.preventDefault(); finish(); } }, { passive: false });
    surface.addEventListener('touchstart', ev => { if (state.pinch.safariGesture || ev.touches.length !== 2) return; const d = distance(ev.touches); if (begin('touch', d)) ev.preventDefault(); }, { passive: false });
    surface.addEventListener('touchmove', ev => { if (!state.pinch.active || state.pinch.source !== 'touch' || ev.touches.length !== 2) return; ev.preventDefault(); move(distance(ev.touches) / Math.max(1, state.pinch.startDistance)); }, { passive: false });
    surface.addEventListener('touchend', ev => { if (state.pinch.active && state.pinch.source === 'touch' && ev.touches.length < 2) finish(); }, { passive: true });
  }

  async function setMode(mode, { recapture = true, broadcast = false, remote = false } = {}) {
    if (!PROFILES[mode]) return;
    if (!state.host && !remote && state.room) return; // guest follows host
    state.mode = mode;
    $$('.mode-btn').forEach(b => b.classList.toggle('active', b.dataset.mode === mode));
    if (ui.xtremeCard) ui.xtremeCard.hidden = mode !== 'xtreme';
    if (ui.qualityHint) ui.qualityHint.textContent = mode === 'xtreme' ? 'Máximo potencial disponible' : 'Adaptativa y estable';
    if (ui.callXtremeBtn) ui.callXtremeBtn.classList.toggle('active', mode === 'xtreme');
    if (ui.requestedQuality) ui.requestedQuality.textContent = mode === 'xtreme' ? 'EXTREMO · máximo disponible' : 'AUTO · adaptativa';
    if (state.view === 'prejoin') {
      state.directOnly = mode === 'xtreme' ? true : !!ui.directOnlyToggle?.checked;
      if (ui.directOnlyToggle) { ui.directOnlyToggle.checked = state.directOnly; ui.directOnlyToggle.disabled = mode === 'xtreme' || !state.host; }
    }
    syncInviteUrl();
    if (recapture && state.localStream) {
      const oldVideo = currentVideoTrack();
      try {
        const { stream } = await acquireMedia({ facing: state.facing, deviceId: state.lastDevice[state.facing], preserveAudio: true });
        const newVideo = stream.getVideoTracks()[0];
        await replacePublishedVideoTrack(newVideo, oldVideo);
        if (oldVideo && oldVideo !== newVideo) oldVideo.stop();
      } catch (err) { console.warn('No se pudo recapturar al cambiar de modo', err); }
      if (state.active) await tuneSenders();
    }
    if (broadcast && state.active && state.sdk?.sendData) {
      try { state.sdk.sendData({ type: 'luma-mode', mode }); } catch {}
    }
  }

    }
  }

  async function loadSdk() {
    if (typeof window.VDONinjaSDK === 'function') return true;
    const sources = [
      'https://cdn.jsdelivr.net/npm/@vdoninja/sdk@1.5.5/vdoninja-sdk.min.js',
      'https://unpkg.com/@vdoninja/sdk@1.5.5/vdoninja-sdk.min.js',
      'https://cdn.jsdelivr.net/gh/steveseguin/ninjasdk@latest/vdoninja-sdk.min.js'
    ];
    for (const src of sources) {
      try {
        await new Promise((resolve, reject) => {
          const script = document.createElement('script'); let done = false;
          const timer = setTimeout(() => { if (!done) { done = true; script.remove(); reject(new Error('timeout')); } }, 8000);
          script.src = src; script.async = true;
          script.onload = () => { if (!done) { done = true; clearTimeout(timer); resolve(); } };
          script.onerror = () => { if (!done) { done = true; clearTimeout(timer); script.remove(); reject(new Error('load error')); } };
          document.head.appendChild(script);
        });
        if (typeof window.VDONinjaSDK === 'function') return true;
